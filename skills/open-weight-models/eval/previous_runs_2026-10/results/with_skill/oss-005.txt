# DBTITLE 1,Idempotent Cleanup
# ── Idempotent cleanup ───────────────────────────────────────────────────
# First run: remove any pre-existing endpoint + UC model versions.
# Subsequent runs: if the endpoint is already READY, skip cleanup and skip rebuild.
# Tear down only when the endpoint is missing or in a failed / broken state.

import time
import mlflow
from mlflow import MlflowClient
from databricks.sdk import WorkspaceClient

w = WorkspaceClient()
mlflow.set_tracking_uri("databricks")
mlflow.set_registry_uri("databricks-uc")
ml_client = MlflowClient()


def _enum_value(value):
    return getattr(value, "value", value)


def _served_entities(endpoint):
    out = []
    for cfg in [getattr(endpoint, "pending_config", None), getattr(endpoint, "config", None)]:
        if cfg and getattr(cfg, "served_entities", None):
            out.extend(cfg.served_entities)
    return out


def _endpoint_is_broken(endpoint):
    ready_state = _enum_value(getattr(getattr(endpoint, "state", None), "ready", None))
    if ready_state == "READY":
        return False
    for se in _served_entities(endpoint):
        dep_state = _enum_value(getattr(getattr(se, "state", None), "deployment", None))
        if dep_state == "DEPLOYMENT_FAILED":
            return True
    return True


def _wait_endpoint_deleted(name, timeout_s=900):
    start = time.time()
    while time.time() - start < timeout_s:
        try:
            w.serving_endpoints.get(name)
            time.sleep(10)
        except Exception:
            return
    raise TimeoutError(f"Timed out waiting for endpoint deletion: {name}")


SKIP_REBUILD_AND_DEPLOY = False
EXISTING_READY_VERSION = None
existing_endpoint = None

try:
    existing_endpoint = w.serving_endpoints.get(ENDPOINT_NAME)
    ready_state = _enum_value(getattr(existing_endpoint.state, "ready", None))
    config_update_state = _enum_value(getattr(existing_endpoint.state, "config_update", None))
    print(f"Found endpoint {ENDPOINT_NAME}: ready={ready_state}, config_update={config_update_state}")

    if ready_state == "READY" and config_update_state == "NOT_UPDATING":
        SKIP_REBUILD_AND_DEPLOY = True
        if existing_endpoint.config and existing_endpoint.config.served_entities:
            EXISTING_READY_VERSION = str(existing_endpoint.config.served_entities[0].entity_version)
        print(f"Endpoint already READY on version {EXISTING_READY_VERSION}; skipping cleanup and rebuild.")
    elif _endpoint_is_broken(existing_endpoint):
        print("Endpoint is not READY; deleting before a clean redeploy.")
        w.serving_endpoints.delete(ENDPOINT_NAME)
        _wait_endpoint_deleted(ENDPOINT_NAME)
        existing_endpoint = None
    else:
        print("Endpoint exists but is not yet stable; deleting for a clean rerun.")
        w.serving_endpoints.delete(ENDPOINT_NAME)
        _wait_endpoint_deleted(ENDPOINT_NAME)
        existing_endpoint = None
except Exception as exc:
    print(f"Endpoint {ENDPOINT_NAME!r} does not exist or could not be read cleanly: {exc}")
    existing_endpoint = None

if not SKIP_REBUILD_AND_DEPLOY:
    existing_versions = list(ml_client.search_model_versions(f"name = '{UC_MODEL_NAME}'"))
    if existing_versions:
        print(f"Deleting {len(existing_versions)} existing model version(s) for {UC_MODEL_NAME} ...")
        for mv in existing_versions:
            print(f"  deleting version {mv.version}")
            ml_client.delete_model_version(name=UC_MODEL_NAME, version=mv.version)
    else:
        print("No model versions found to delete.")
else:
    print("Cleanup skipped because endpoint is already READY.")

print(f"SKIP_REBUILD_AND_DEPLOY = {SKIP_REBUILD_AND_DEPLOY}")
print(f"EXISTING_READY_VERSION = {EXISTING_READY_VERSION}")

# DBTITLE 1,Download Geneformer Snapshot
# ── Download Geneformer from Hugging Face and copy to Volume ─────────────
# HLS refs:
#   * Download to local temp, then copy to UC Volume.
#   * Use sentinel files for idempotent reruns.
#   * Disable HF Xet before huggingface_hub import by using a subprocess.

import json
import os
import pickle
import shutil
import subprocess
import sys
import textwrap
from pathlib import Path


def discover_token_dict(snapshot_root: str) -> str:
    candidates = [
        EXPECTED_TOKEN_DICT,
        os.path.join(snapshot_root, "geneformer", "token_dictionary_gc30M.pkl"),
    ]
    for candidate in candidates:
        if os.path.exists(candidate):
            return candidate
    for root, _, files in os.walk(snapshot_root):
        for file_name in files:
            if "token_dictionary" in file_name.lower() and file_name.endswith(".pkl"):
                return os.path.join(root, file_name)
    raise FileNotFoundError(f"Could not find token dictionary under {snapshot_root}")


if SKIP_REBUILD_AND_DEPLOY and os.path.exists(COPY_SENTINEL):
    print(f"Sentinel present and endpoint READY; reusing artifacts at {SNAPSHOT_ROOT}")
else:
    if os.path.exists(LOCAL_SNAPSHOT_DIR):
        shutil.rmtree(LOCAL_SNAPSHOT_DIR)
    os.makedirs(LOCAL_SNAPSHOT_DIR, exist_ok=True)

    download_script = textwrap.dedent(
        """
        import os
        os.environ["HF_HUB_DISABLE_XET"] = "1"
        from huggingface_hub import snapshot_download
        snapshot_download(
            repo_id=os.environ["HF_REPO"],
            local_dir=os.environ["LOCAL_SNAPSHOT_DIR"],
            local_dir_use_symlinks=False,
            resume_download=True,
        )
        """
    )
    env = dict(os.environ)
    env.update({
        "HF_REPO": HF_REPO,
        "LOCAL_SNAPSHOT_DIR": LOCAL_SNAPSHOT_DIR,
        "HF_HUB_DISABLE_XET": "1",
    })

    print(f"Downloading {HF_REPO} into {LOCAL_SNAPSHOT_DIR} ...")
    subprocess.run([sys.executable, "-c", download_script], check=True, env=env)

    if os.path.exists(SNAPSHOT_ROOT):
        shutil.rmtree(SNAPSHOT_ROOT)
    shutil.copytree(LOCAL_SNAPSHOT_DIR, SNAPSHOT_ROOT, dirs_exist_ok=True)

    Path(COPY_SENTINEL).write_text(datetime.now(timezone.utc).isoformat())
    print(f"Snapshot copied to {SNAPSHOT_ROOT}")

# DBTITLE 1,Geneformer PyFunc Wrapper Source
# ── Geneformer PyFunc wrapper source ─────────────────────────────────────
# HLS refs enforced here:
#   * file-based MLflow logging, not cloudpickle
#   * explicit imports inside the wrapper file
#   * isatty patch before from_pretrained
#   * real token dictionary + bounded tabular serving contract

import textwrap

WRAPPER_FILE_CONTENT = textwrap.dedent(
    '''
    import io
    import json
    import os
    import pickle
    import sys
    from contextlib import nullcontext

    import mlflow
    import numpy as np
    import pandas as pd
    import torch
    from transformers import BertModel


    class GeneformerEmbedder(mlflow.pyfunc.PythonModel):
        def load_context(self, context):
            for stream in (sys.stdout, sys.stderr):
                if stream is not None and not hasattr(stream, "isatty"):
                    stream.isatty = lambda: False

            self.checkpoint_dir = context.artifacts["checkpoint_dir"]
            self.token_dict_path = context.artifacts["token_dict"]
            self.vocab_version = "v1"

            with open(self.token_dict_path, "rb") as handle:
                self.token_dict = pickle.load(handle)

            self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
            self.model = BertModel.from_pretrained(
                self.checkpoint_dir,
                local_files_only=True,
                add_pooling_layer=False,
            )
            self.model.eval()
            self.model.to(self.device)
            self.hidden_size = int(getattr(self.model.config, "hidden_size", 256))

        def _decode_json_array(self, value):
            if isinstance(value, str):
                return json.loads(value)
            return value

        def _parse_config(self, raw_value):
            defaults = {
                "truncation": True,
                "gene_count_limit": "2048",
                "pooling_mode": "mean",
            }
            if raw_value is None or (isinstance(raw_value, float) and np.isnan(raw_value)):
                return defaults
            if isinstance(raw_value, str):
                parsed = json.loads(raw_value)
            elif isinstance(raw_value, dict):
                parsed = raw_value
            else:
                raise TypeError(f"Unsupported config type: {type(raw_value)}")
            defaults.update(parsed)
            return defaults

        def _normalize_row(self, row_dict, params=None):
            genes = self._decode_json_array(row_dict["genes"])
            expression = self._decode_json_array(row_dict["expression"])
            config = self._parse_config(row_dict.get("config"))
            if params:
                config.update({k: v for k, v in params.items() if v is not None})

            if len(genes) != len(expression):
                raise ValueError("genes and expression must have the same length")

            valid_pairs = []
            for gene, expr in zip(genes, expression):
                if gene not in self.token_dict:
                    continue
                expr = float(expr)
                if expr <= 0:
                    continue
                valid_pairs.append((gene, expr))

            if not valid_pairs:
                raise ValueError("No valid positive-expression genes were present in the Geneformer vocabulary")

            valid_pairs.sort(key=lambda item: item[1], reverse=True)
            if bool(config.get("truncation", True)):
                limit = int(config.get("gene_count_limit", "2048"))
                valid_pairs = valid_pairs[:limit]

            token_ids = [int(self.token_dict[gene]) for gene, _ in valid_pairs]
            return token_ids, str(config.get("pooling_mode", "mean")).lower()

        def _embed(self, token_ids, pooling_mode):
            input_ids = torch.tensor([token_ids], dtype=torch.long, device=self.device)
            attention_mask = torch.ones_like(input_ids)
            autocast_ctx = (
                torch.autocast(device_type="cuda", dtype=torch.bfloat16)
                if self.device.type == "cuda"
                else nullcontext()
            )
            with torch.no_grad():
                with autocast_ctx:
                    outputs = self.model(input_ids=input_ids, attention_mask=attention_mask)
                    hidden = outputs.last_hidden_state
                    if pooling_mode in {"mean", "avg"}:
                        pooled = hidden.mean(dim=1)
                    elif pooling_mode in {"cls", "position0"}:
                        pooled = hidden[:, 0, :]
                    else:
                        raise ValueError(f"Unsupported pooling_mode: {pooling_mode}")
            return pooled[0].detach().float().cpu().numpy().tolist()

        def predict(self, context, model_input, params=None):
            if isinstance(model_input, dict):
                model_input = pd.DataFrame(model_input)
            elif not isinstance(model_input, pd.DataFrame):
                model_input = pd.DataFrame(model_input)

            outputs = []
            for _, row in model_input.iterrows():
                row_dict = row.to_dict()
                token_ids, pooling_mode = self._normalize_row(row_dict, params=params)
                embedding = self._embed(token_ids, pooling_mode)
                outputs.append({
                    "cell_id": str(row_dict.get("cell_id", "unknown")),
                    "embedding": embedding,
                    "embedding_dim": len(embedding),
                    "vocab_version": str(row_dict.get("vocab_version", self.vocab_version)),
                })
            return pd.DataFrame(outputs)


    mlflow.models.set_model(GeneformerEmbedder())
    '''
)

print(f"Wrapper source prepared ({len(WRAPPER_FILE_CONTENT)} chars)")

import json

    AiGatewayUsageTrackingConfig,

w = WorkspaceClient()

def _enum_value(value):
    return getattr(value, "value", value)


def _get_events(endpoint_name: str):
    try:
        return w.api_client.do("GET", f"/api/2.0/serving-endpoints/{endpoint_name}/events")
    except Exception as exc:
        return {"error": str(exc)}


def _get_logs(endpoint_name: str, served_entity_name: str, config_version: int | None):
    try:
        query = {}
        if config_version is not None:
            query["config_version"] = config_version
        return w.api_client.do(
            "GET",
            f"/api/2.0/serving-endpoints/{endpoint_name}/served-entities/{served_entity_name}/logs",
            query=query,
        )
    except Exception as exc:
        return {"error": str(exc)}


def _wait_ready(endpoint_name: str, target_version: str, timeout_s: int = 1800):
    start = time.time()
    while time.time() - start < timeout_s:
        ep = w.serving_endpoints.get(endpoint_name)
        for cfg in [getattr(ep, "pending_config", None), getattr(ep, "config", None)]:
            if cfg and getattr(cfg, "served_entities", None):
                for se in cfg.served_entities:
                    dep_state = _enum_value(getattr(getattr(se, "state", None), "deployment", None))
                    if dep_state == "DEPLOYMENT_FAILED":
                        message = getattr(getattr(se, "state", None), "deployment_state_message", "DEPLOYMENT_FAILED")
                        config_version = getattr(cfg, "config_version", None)
                        return False, ep, {
                            "deployment_state_message": message,
                            "config_version": config_version,
                            "events": _get_events(endpoint_name),
                            "logs": _get_logs(endpoint_name, served_entity_name=se.name, config_version=config_version),
                        }

        ready_state = _enum_value(getattr(ep.state, "ready", None))
        config_update_state = _enum_value(getattr(ep.state, "config_update", None))
        active_version = None
        if ep.config and ep.config.served_entities:
            active_version = str(ep.config.served_entities[0].entity_version)

        print(f"Polling endpoint: ready={ready_state}, config_update={config_update_state}, active_version={active_version}")
        if ready_state == "READY" and config_update_state == "NOT_UPDATING" and active_version == target_version:
            return True, ep, None
        time.sleep(30)

    raise TimeoutError(f"Timed out waiting for endpoint {endpoint_name} to become READY")


if not RUN_GO:
    print("run_go=false; skipping endpoint deployment.")
else:
    existing_endpoint = None
    try:
        existing_endpoint = w.serving_endpoints.get(ENDPOINT_NAME)
    except Exception:
        existing_endpoint = None

    if SKIP_REBUILD_AND_DEPLOY and existing_endpoint is not None:
        print(f"Endpoint {ENDPOINT_NAME} already READY on version {TARGET_MODEL_VERSION}; skipping create/update.")
    elif existing_endpoint is not None:
        print(f"Found existing endpoint {ENDPOINT_NAME}; resuming poll instead of creating again.")
        ok, endpoint_info, failure_info = _wait_ready(ENDPOINT_NAME, TARGET_MODEL_VERSION)
        if not ok:
            print("Deployment failed while resuming poll; captured diagnostics:")
            print(json.dumps(failure_info, indent=2)[:12000])
            raise RuntimeError(f"Endpoint deployment failed after resume: {json.dumps(failure_info)[:12000]}")
        print(f"Endpoint READY on version {TARGET_MODEL_VERSION}")
    else:
        last_failure = None
        for attempt in range(1, 3):
            table_prefix = f"{ENDPOINT_NAME}_{datetime.now():%Y%m%d%H%M%S}_{attempt}"
            ai_gateway = AiGatewayConfig(
                inference_table_config=AiGatewayInferenceTableConfig(
                    enabled=True,
                    catalog_name=CATALOG,
                    schema_name=SCHEMA,
                    table_name_prefix=table_prefix,
                ),
                usage_tracking_config=AiGatewayUsageTrackingConfig(enabled=True),
            )
            endpoint_config = EndpointCoreConfigInput(
                name=ENDPOINT_NAME,
                served_entities=[
                    ServedEntityInput(
                        name=SERVED_ENTITY_NAME,
                        entity_name=UC_MODEL_NAME,
                        entity_version=TARGET_MODEL_VERSION,
                        workload_type=ServingModelWorkloadType.GPU_SMALL,
                        workload_size="Small",
                        scale_to_zero_enabled=True,
                    )
                ],
            )

            print(f"Deployment attempt {attempt}: model version {TARGET_MODEL_VERSION}, inference table prefix {table_prefix}")
            w.serving_endpoints.create(name=ENDPOINT_NAME, config=endpoint_config, ai_gateway=ai_gateway)
            ok, endpoint_info, failure_info = _wait_ready(ENDPOINT_NAME, TARGET_MODEL_VERSION)
            if ok:
                print(f"Endpoint READY on version {TARGET_MODEL_VERSION}")
                break

            last_failure = failure_info
            print("Deployment failed; captured diagnostics:")
            print(json.dumps(failure_info, indent=2)[:12000])

            try:
                w.serving_endpoints.delete(ENDPOINT_NAME)
                start = time.time()
                while time.time() - start < 900:
                    try:
                        w.serving_endpoints.get(ENDPOINT_NAME)
                        time.sleep(10)
                    except Exception:
                        break
            except Exception as exc:
                print(f"Cleanup after failed deploy raised: {exc}")
        else:
            raise RuntimeError(f"Endpoint deployment failed after retries: {json.dumps(last_failure)[:12000]}")
